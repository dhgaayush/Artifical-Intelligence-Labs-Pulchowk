\section{Theoretical Background}

\subsection{First Order Predicate Logic}
First Order Predicate Logic (FOPL) extends propositional logic by introducing predicates, quantifiers, and variables. It provides a formal framework for representing knowledge about objects and their relationships. The key components are: predicates (e.g.\ $\text{Horse}(x)$), constants (e.g.\ $charlie$), variables (e.g.\ $X$), universal quantifier ($\forall$: ``for all''), and existential quantifier ($\exists$: ``there exists'').

\subsection{FOPL to Prolog Mapping}
Prolog is a computational realisation of Horn-clause logic. Universal quantifiers become Prolog variables, implications become rules (\texttt{:-}), conjunctions become comma-separated sub-goals, and ground facts represent existential or specific assertions.

\subsection{Inference Strategies}
\textbf{Backward chaining} (Prolog's default) starts from the goal and works backwards to facts via unification and backtracking. \textbf{Forward chaining} starts from known facts and applies rules iteratively to derive new conclusions. Both are sound and complete for Horn-clause knowledge bases.

\section{Assignment 1 --- Course Preference via Resolution}

\subsection{Problem Summary}
A knowledge base describes Steve's course preferences. Steve only likes easy courses; science courses are hard; basket weaving courses are easy; and BK301 is a basket weaving course. The task is to use resolution to determine which course Steve would like.

\subsection{FOPL Representation}
\begin{foplbox}
\begin{align*}
  1.\quad & \forall X\;(\text{course}(X) \land \text{easy}(X) \Rightarrow \text{likes}(steve,X)) \\
  2.\quad & \forall X\;(\text{science\_course}(X) \Rightarrow \text{hard}(X)) \\
  3.\quad & \forall X\;(\text{basket\_weaving\_course}(X) \Rightarrow \text{easy}(X)) \\
  4.\quad & \text{basket\_weaving\_course}(bk301) \\
  5.\quad & \text{course}(bk301)
\end{align*}
\end{foplbox}

\subsection{Resolution Proof}
\begin{enumerate}
  \item $\text{basket\_weaving\_course}(bk301)$ --- given fact.
  \item By rule (3): $\text{easy}(bk301)$.
  \item $\text{course}(bk301)$ --- given fact.
  \item By rule (1): $\text{course}(bk301) \land \text{easy}(bk301) \Rightarrow \text{likes}(steve, bk301)$.
  \item $\therefore\; \text{likes}(steve, bk301)$. $\blacksquare$
\end{enumerate}

\subsection{Prolog Code}
\begin{lstlisting}[style=prolog, caption={Course preference resolution}]
course(bk301).
basket_weaving_course(bk301).

likes(steve, X) :-
    course(X),
    easy(X).

hard(X) :-
    science_course(X).

easy(X) :-
    basket_weaving_course(X).
\end{lstlisting}

\subsection{Output}
\begin{lstlisting}[style=output, caption={Assignment 1 output}]
?- likes(steve, X).
X = bk301.

?- likes(steve, bk301).
true.
\end{lstlisting}

\subsection{Discussion}
Resolution chains the basket-weaving fact through the ``easy'' rule and combines it with the ``course'' fact to satisfy Steve's preference condition. Prolog returns \texttt{X = bk301} via automatic backward chaining and unification.

\section{Assignment 2 --- Horses, Mammals, and Inverse Relations}

\subsection{Problem Summary}
Five statements about horses, mammals, offspring, and parent relations are given. The goal is to determine whether Charlie is a horse, using the facts that Bluebeard is Charlie's parent and Bluebeard is a horse.

\subsection{FOPL Representation}
\begin{foplbox}
\begin{align*}
  1.\quad & \forall X\;(\text{Horse}(X) \Rightarrow \text{Mammal}(X)) \\
  2.\quad & \forall X,Y\;(\text{Offspring}(X,Y) \land \text{Horse}(Y) \Rightarrow \text{Horse}(X)) \\
  3.\quad & \text{Parent}(bluebeard, charlie) \\
  4.\quad & \forall X,Y\;(\text{Offspring}(X,Y) \Leftrightarrow \text{Parent}(Y,X)) \\
  5.\quad & \forall X\;(\text{Mammal}(X) \Rightarrow \exists Y\;\text{Parent}(Y,X))
\end{align*}
\end{foplbox}

\subsection{Prolog Code}
\begin{lstlisting}[style=prolog, caption={Horses and mammals with inverse relations}]
parent(bluebeard, charlie).
horse(bluebeard).

mammal(X) :- horse(X).

horse(X) :-
    offspring(X, Y),
    horse(Y).

offspring(X, Y) :- parent(Y, X).
\end{lstlisting}

\subsection{Inference Trace}
\begin{tracebox}
\begin{verbatim}
?- horse(charlie).
  Goal: horse(charlie)
    offspring(charlie, Y) -> parent(Y, charlie) -> Y = bluebeard  [TRUE]
    horse(bluebeard)  [FACT -> TRUE]
  => TRUE

?- mammal(charlie).
  mammal(charlie) <- horse(charlie) [TRUE] => TRUE
\end{verbatim}
\end{tracebox}

\subsection{Output}
\begin{lstlisting}[style=output, caption={Assignment 2 output}]
?- horse(charlie).
true.

?- mammal(charlie).
true.

?- offspring(charlie, bluebeard).
true.
\end{lstlisting}

\subsection{Discussion}
Charlie is the offspring of Bluebeard (inverse of parent relation). Since Bluebeard is a horse, Charlie inherits the horse property via rule (2). Consequently, Charlie is also a mammal via rule (1). This demonstrates inverse relations and transitive rule chaining in Prolog.

\section{Assignment 3 --- Student Evaluation: Forward \& Backward Chaining}

\subsection{Problem Summary}
A Student Evaluation System defines three rules linking lecture attendance, studying, course coverage, good position, and first position. Given that a student attends all lectures, studies course books, and has no one with higher marks, the task is to prove the student will get first position using both forward and backward chaining.

\subsection{FOPL Representation}
\begin{foplbox}
\begin{align*}
  \text{Rule I:}\quad  & \forall X\;(\text{GoodPosition}(X) \land \text{NoHigherMarks}(X) \Rightarrow \text{FirstPosition}(X)) \\
  \text{Rule II:}\quad & \forall X\;(\text{AttendsLectures}(X) \land \text{StudiesBooks}(X) \Rightarrow \text{CoversCourse}(X)) \\
  \text{Rule III:}\quad& \forall X\;(\text{CoversCourse}(X) \Rightarrow \text{GoodPosition}(X))
\end{align*}
\textbf{Facts:}\; $\text{AttendsLectures}(student),\; \text{StudiesBooks}(student),\; \text{NoHigherMarks}(student)$
\end{foplbox}

\subsection{Prolog Code (Backward Chaining)}
\begin{lstlisting}[style=prolog, caption={Student evaluation --- backward chaining}]
attends_lectures(student).
studies_books(student).
no_higher_marks(student).

covers_course(X) :-
    attends_lectures(X),
    studies_books(X).

good_position(X) :-
    covers_course(X).

first_position(X) :-
    good_position(X),
    no_higher_marks(X).
\end{lstlisting}

\subsection{Backward Chaining Trace}
\begin{tracebox}
\begin{verbatim}
Goal: first_position(student)
  |- good_position(student) AND no_higher_marks(student)
  |     |                            |
  |     |                            +-> FACT => TRUE
  |     +-> covers_course(student)
  |              +-> attends_lectures(student)  [FACT]
  |              +-> studies_books(student)      [FACT]
  |           => TRUE
  => TRUE   (Q.E.D.)
\end{verbatim}
\end{tracebox}

\subsection{Forward Chaining Trace}
\begin{tracebox}
\begin{tabular}{@{}clp{8cm}@{}}
\toprule
\textbf{Step} & \textbf{Rule} & \textbf{Derivation} \\
\midrule
1 & Rule II &
  $\text{attends\_lectures}(student) \land \text{studies\_books}(student)$
  $\Rightarrow$ \textbf{covers\_course(student)} \\[6pt]
2 & Rule III &
  $\text{covers\_course}(student)$
  $\Rightarrow$ \textbf{good\_position(student)} \\[6pt]
3 & Rule I &
  $\text{good\_position}(student) \land \text{no\_higher\_marks}(student)$
  $\Rightarrow$ \textbf{first\_position(student)} $\blacksquare$ \\
\bottomrule
\end{tabular}
\end{tracebox}

\subsection{Forward Chaining in Prolog}
\begin{lstlisting}[style=prolog, caption={Forward chaining using assert/1}]
:- dynamic derived/1.

forward_chain :-
    (   attends_lectures(S), studies_books(S),
        \+ derived(covers_course(S))
    ->  assert(derived(covers_course(S))),
        format('Step 1 (Rule II):  Derived covers_course(~w)~n', [S])
    ;   true),
    (   derived(covers_course(S)),
        \+ derived(good_position(S))
    ->  assert(derived(good_position(S))),
        format('Step 2 (Rule III): Derived good_position(~w)~n', [S])
    ;   true),
    (   derived(good_position(S)), no_higher_marks(S),
        \+ derived(first_position(S))
    ->  assert(derived(first_position(S))),
        format('Step 3 (Rule I):   Derived first_position(~w)~n', [S])
    ;   true).
\end{lstlisting}

\subsection{Output}
\begin{lstlisting}[style=output, caption={Assignment 3 output}]
?- first_position(student).
true.

?- forward_chain.
Step 1 (Rule II):  Derived covers_course(student)
Step 2 (Rule III): Derived good_position(student)
Step 3 (Rule I):   Derived first_position(student)
true.
\end{lstlisting}

\subsection{Discussion}
Both inference strategies reach the same conclusion. Backward chaining decomposes the goal top-down into sub-goals resolved against facts. Forward chaining starts from facts and derives new conclusions bottom-up until the goal is reached. Both confirm: \textbf{the student will get first position}.
